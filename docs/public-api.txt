list.more	List	Load the next page
history.fold	History	Fold a group open or shut
history.expand-all	History	Expand every group
history.collapse-all	History	Collapse every group
history.plumbing	History	Show/hide workflow tasks
history.timeline	History	Timeline, like the web UI
history.gaps	History	Fold/unfold idle time
history.next-failure	History	Jump to the next failure
history.prev-failure	History	Jump to the previous failure
history.follow	History	Follow, tail a running workflow
history.detail	History	Show the payloads under the cursor

history.detail-up	History	Scroll the payload pane up

payload.edit	History	Open the payloads in $EDITOR
history.source	History	Go to the source of this event
history.child	History	Open the child workflow
history.parent	History	Open the parent workflow
window.split-right	Windows	Split side by side
window.split-down	Windows	Split above and below
window.close	Windows	Close this window

window.focus-left	Windows	Focus the window left
window.focus-right	Windows	Focus the window right
window.focus-up	Windows	Focus the window above
window.focus-down	Windows	Focus the window below
window.grow-left	Windows	Widen to the left
window.grow-right	Windows	Widen to the right
window.grow-up	Windows	Grow upwards
window.grow-down	Windows	Grow downwards
workflow.cancel	Mutate	Cancel this workflow
workflow.terminate	Mutate	Terminate this workflow
workflow.signal	Mutate	Signal this workflow
workflow.delete	Mutate	Delete this workflow
workflow.reset	Mutate	Reset to the event here
workflow.update	Mutate	Send an update
schedule.pause	Mutate	Pause or resume a schedule
schedule.trigger	Mutate	Run a schedule now
schedule.delete	Mutate	Delete this schedule
schedule.backfill	Mutate	Backfill a time range
schedule.create	Mutate	Create a schedule
workflow.query	Query	Ask the workflow a query
workflow.stack	Query	Where the workflow is blocked

tab.close	Tabs	Close this tab

tab.previous	Tabs	Previous tab
